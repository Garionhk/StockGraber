  componentDidMount(){ this.scheduleDraw(); this._onResize=()=>this.scheduleDraw(); window.addEventListener('resize',this._onResize); [60,200,500].forEach(t=>setTimeout(()=>this.scheduleDraw(),t)); if(document.fonts&&document.fonts.ready) document.fonts.ready.then(()=>this.scheduleDraw()); }
  componentDidUpdate(){ this.scheduleDraw(); }
  componentWillUnmount(){ window.removeEventListener('resize',this._onResize); if(this._ro)this._ro.disconnect(); }

  redrawAll(){
    const cv=this.cv; const stack=cv.stackEl; if(!stack) return;
    const rect=stack.getBoundingClientRect(); const W=rect.width; if(W<2) return;
    this._up=this.props.upColor||'#22c98a'; this._dn=this.props.downColor||'#f6465d'; this._accent=this.props.accent||'#a855f7';
    const {bars,hs,N}=this.data; const n=this.state.viewN; const start=Math.max(0,N-n);
    const view=bars.slice(start);
    const closes=bars.map(b=>b.c);
    const parts=this.state.preset.split('/').map(Number); const fp=parts[0], sp=parts[1];
    const maFn=this.state.ema?this.ema:this.sma;
    const maF=maFn.call(this,closes,fp).slice(start);
    const maS=maFn.call(this,closes,sp).slice(start);
    const rsi=this.rsiArr(closes,14).slice(start);
    const mc=this.macdArr(closes); const mM=mc.m.slice(start),mS=mc.sg.slice(start),mH=mc.hist.slice(start);
    const hsV=hs.slice(start); const baseR=view.length?view[0].c/hsV[0]:1; const cmp=hsV.map(v=>v*baseR);
    const padR=58; const m=view.length; const xStep=(W-padR)/Math.max(1,m);
    this.geom={xStep,padR,n:m,W,start};
    const monthIdx=[]; let pm=null; view.forEach((b,i)=>{const mm=b.d.getUTCMonth();if(mm!==pm){pm=mm;monthIdx.push(i);}});
    const bw=Math.max(1,Math.min(13,xStep*0.62));
    const bottom = this.state.macd?'macd':(this.state.rsi?'rsi':(this.state.vol?'vol':'price'));
    const g={view,maF,maS,rsi,mM,mS,mH,cmp,xStep,padR,bw,monthIdx};
    if(cv.price) this.drawPrice(cv.price,g,bottom==='price');
    if(cv.volume&&this.state.vol) this.drawVolume(cv.volume,g,bottom==='vol');
    if(cv.rsi&&this.state.rsi) this.drawRSI(cv.rsi,g,bottom==='rsi');
    if(cv.macd&&this.state.macd) this.drawMACD(cv.macd,g,bottom==='macd');
  }

  vgrid(x,h,monthIdx,xStep){ x.strokeStyle='rgba(255,255,255,0.04)'; x.lineWidth=1; monthIdx.forEach(i=>{const cx=Math.round(i*xStep+xStep/2)+0.5; x.beginPath();x.moveTo(cx,0);x.lineTo(cx,h);x.stroke();}); }
  dates(x,h,view,monthIdx,xStep){ x.fillStyle='#5a606b'; x.font='11px "JetBrains Mono", monospace'; x.textAlign='center'; x.textBaseline='alphabetic'; monthIdx.forEach(i=>{const cx=i*xStep+xStep/2; x.fillText(this.monthLabel(view[i].d),cx,h-7);}); }

  drawPrice(c,g,showDates){
    const {x,w,h}=this.fit(c); const padT=12,padB=showDates?24:10;
    const {view,maF,maS,cmp,xStep,padR,bw,monthIdx}=g;
    let lo=Infinity,hi=-Infinity; view.forEach(b=>{if(b.l<lo)lo=b.l;if(b.h>hi)hi=b.h;});
    [maF,maS].forEach(a=>a.forEach(v=>{if(v!=null){if(v<lo)lo=v;if(v>hi)hi=v;}}));
    if(this.state.compare) cmp.forEach(v=>{if(v<lo)lo=v;if(v>hi)hi=v;});
    const lg=this.state.log; const tf=v=>lg?Math.log(v):v; const ylo=tf(lo),yhi=tf(hi); const pdf=(yhi-ylo)*0.06||1;
    const A=ylo-pdf,B=yhi+pdf; const Y=v=>padT+(1-(tf(v)-A)/(B-A))*(h-padT-padB);
    x.clearRect(0,0,w,h);
    for(let k=0;k<=5;k++){ const v=lo+(hi-lo)*k/5; const yy=Math.round(Y(v))+0.5; x.strokeStyle='rgba(255,255,255,0.045)';x.lineWidth=1;x.beginPath();x.moveTo(0,yy);x.lineTo(w-padR,yy);x.stroke(); x.fillStyle='#6a707a';x.font='11px "JetBrains Mono", monospace';x.textAlign='left';x.textBaseline='middle';x.fillText(v.toFixed(0),w-padR+8,Y(v)); }
    this.vgrid(x,h,monthIdx,xStep);
    view.forEach((b,i)=>{const cx=i*xStep+xStep/2;const up=b.c>=b.o;const col=up?this._up:this._dn;x.strokeStyle=col;x.fillStyle=col;x.lineWidth=1;x.beginPath();x.moveTo(Math.round(cx)+0.5,Y(b.h));x.lineTo(Math.round(cx)+0.5,Y(b.l));x.stroke();const yo=Y(b.o),yc=Y(b.c);let top=Math.min(yo,yc),hh=Math.abs(yc-yo);if(hh<1.2)hh=1.2;x.fillRect(cx-bw/2,top,bw,hh);});
    const line=(arr,color,wd)=>{x.strokeStyle=color;x.lineWidth=wd;x.beginPath();let st=false;arr.forEach((v,i)=>{if(v==null){st=false;return;}const cx=i*xStep+xStep/2,yy=Y(v);if(!st){x.moveTo(cx,yy);st=true;}else x.lineTo(cx,yy);});x.stroke();};
    line(maS,'#f3a13a',1.5); line(maF,'#4f8cff',1.5);
    if(this.state.compare) line(cmp,'#22d3ee',1.4);
    let lastCx=null,lastCy=null;
    for(let i=1;i<view.length;i++){ if(maF[i]==null||maS[i]==null||maF[i-1]==null||maS[i-1]==null)continue; const pr=maF[i-1]-maS[i-1],cu=maF[i]-maS[i]; const cx=i*xStep+xStep/2; if(pr<=0&&cu>0){this.tri(x,cx,Y(view[i].l)+15,true,this._up);lastCx=cx;lastCy=Y(view[i].l)+15;} else if(pr>=0&&cu<0){this.tri(x,cx,Y(view[i].h)-15,false,this._dn);lastCx=cx;lastCy=Y(view[i].h)-15;} }
    if(lastCx!=null){x.strokeStyle='rgba(234,179,8,0.9)';x.lineWidth=1.5;x.beginPath();x.arc(lastCx,lastCy,12,0,Math.PI*2);x.stroke();}
    if(showDates) this.dates(x,h,view,monthIdx,xStep);
  }
  tri(x,cx,cy,up,color){const s=6;x.fillStyle=color;x.beginPath();if(up){x.moveTo(cx,cy-s);x.lineTo(cx-s,cy+s);x.lineTo(cx+s,cy+s);}else{x.moveTo(cx,cy+s);x.lineTo(cx-s,cy-s);x.lineTo(cx+s,cy-s);}x.closePath();x.fill();}

  drawVolume(c,g,showDates){const {x,w,h}=this.fit(c);const padT=10,padB=showDates?22:8;const {view,xStep,padR,bw,monthIdx}=g;x.clearRect(0,0,w,h);this.vgrid(x,h,monthIdx,xStep);let mv=0;view.forEach(b=>{if(b.v>mv)mv=b.v;});view.forEach((b,i)=>{const cx=i*xStep+xStep/2;const up=b.c>=b.o;x.fillStyle=up?this.hexA(this._up,0.5):this.hexA(this._dn,0.45);const bh=(b.v/mv)*(h-padT-padB);x.fillRect(cx-bw/2,h-padB-bh,bw,bh);});x.fillStyle='#6a707a';x.font='11px "JetBrains Mono", monospace';x.textAlign='left';x.textBaseline='top';x.fillText(this.fmtVol(mv),w-padR+8,padT);x.textBaseline='bottom';x.fillText('0',w-padR+8,h-padB);if(showDates)this.dates(x,h,view,monthIdx,xStep);}

  drawRSI(c,g,showDates){const {x,w,h}=this.fit(c);const padT=10,padB=showDates?22:8;const {rsi,xStep,padR,monthIdx,view}=g;const Y=v=>padT+(1-v/100)*(h-padT-padB);x.clearRect(0,0,w,h);this.vgrid(x,h,monthIdx,xStep);x.fillStyle='rgba(255,255,255,0.028)';x.fillRect(0,Y(70),w-padR,Y(30)-Y(70));[30,70].forEach(v=>{x.strokeStyle='rgba(255,255,255,0.12)';x.setLineDash([4,4]);x.lineWidth=1;x.beginPath();x.moveTo(0,Y(v)+0.5);x.lineTo(w-padR,Y(v)+0.5);x.stroke();x.setLineDash([]);x.fillStyle='#6a707a';x.font='11px "JetBrains Mono", monospace';x.textAlign='left';x.textBaseline='middle';x.fillText(''+v,w-padR+8,Y(v));});x.strokeStyle=this._accent;x.lineWidth=1.5;x.beginPath();let st=false;rsi.forEach((v,i)=>{if(v==null){st=false;return;}const cx=i*xStep+xStep/2,yy=Y(v);if(!st){x.moveTo(cx,yy);st=true;}else x.lineTo(cx,yy);});x.stroke();if(showDates)this.dates(x,h,view,monthIdx,xStep);}

  drawMACD(c,g,showDates){const {x,w,h}=this.fit(c);const padT=12,padB=showDates?22:8;const {mM,mS,mH,xStep,padR,bw,monthIdx,view}=g;let mx=1e-6;[].concat(mM,mS,mH).forEach(v=>{if(v!=null&&Math.abs(v)>mx)mx=Math.abs(v);});mx*=1.15;const Y=v=>padT+(1-(v+mx)/(2*mx))*(h-padT-padB);x.clearRect(0,0,w,h);this.vgrid(x,h,monthIdx,xStep);x.strokeStyle='rgba(255,255,255,0.1)';x.lineWidth=1;x.beginPath();x.moveTo(0,Y(0)+0.5);x.lineTo(w-padR,Y(0)+0.5);x.stroke();mH.forEach((v,i)=>{if(v==null)return;const cx=i*xStep+xStep/2;x.fillStyle=v>=0?this.hexA(this._up,0.55):this.hexA(this._dn,0.5);const y0=Y(0),y1=Y(v);x.fillRect(cx-bw/2,Math.min(y0,y1),bw,Math.abs(y1-y0)||1);});const line=(arr,color)=>{x.strokeStyle=color;x.lineWidth=1.4;x.beginPath();let st=false;arr.forEach((v,i)=>{if(v==null){st=false;return;}const cx=i*xStep+xStep/2,yy=Y(v);if(!st){x.moveTo(cx,yy);st=true;}else x.lineTo(cx,yy);});x.stroke();};line(mM,'#4f8cff');line(mS,'#f3a13a');x.fillStyle='#6a707a';x.font='11px "JetBrains Mono", monospace';x.textAlign='left';x.textBaseline='top';x.fillText(mx.toFixed(1),w-padR+8,padT);x.textBaseline='bottom';x.fillText((-mx).toFixed(1),w-padR+8,h-padB);if(showDates)this.dates(x,h,view,monthIdx,xStep);}

  drawOverlay(){const c=this.cv.overlay;if(!c||!this.geom)return;const {x,w,h}=this.fit(c);x.clearRect(0,0,w,h);const idx=this.hoverIdx;if(idx==null)return;const {xStep}=this.geom;const cx=idx*xStep+xStep/2;x.strokeStyle=this.hexA(this.props.accent||'#a855f7',0.75);x.setLineDash([5,4]);x.lineWidth=1;x.beginPath();x.moveTo(Math.round(cx)+0.5,0);x.lineTo(Math.round(cx)+0.5,h);x.stroke();x.setLineDash([]);}

  updateReadout(idx){const r=this.readout;if(!r.date)return;const N=this.data.N,n=this.state.viewN,start=Math.max(0,N-n);const i= idx==null?(n-1):Math.min(idx,n-1);const b=this.data.bars[start+i];if(!b)return;const pb=this.data.bars[start+i-1]||b;const chg=b.c-pb.c;const pct=pb.c?chg/pb.c*100:0;const up=chg>=0;const UP=this.props.upColor||'#22c98a',DN=this.props.downColor||'#f6465d';r.date.textContent=this.fmtDate(b.d);r.o.textContent=b.o.toFixed(2);r.h.textContent=b.h.toFixed(2);r.l.textContent=b.l.toFixed(2);r.c.textContent=b.c.toFixed(2);r.chg.textContent=(up?'+':'')+chg.toFixed(2)+'  ('+(up?'+':'')+pct.toFixed(2)+'%)';r.chg.style.color=up?UP:DN;if(r.arrow){r.arrow.textContent=up?'▲':'▼';r.arrow.style.color=up?UP:DN;}}

  onMove=(e)=>{const stack=this.cv.stackEl;if(!stack||!this.geom)return;const rect=stack.getBoundingClientRect();const xpos=e.clientX-rect.left;let idx=Math.floor(xpos/this.geom.xStep);idx=Math.max(0,Math.min(this.geom.n-1,idx));this.hoverIdx=idx;this.drawOverlay();this.updateReadout(idx);};
  onLeave=()=>{this.hoverIdx=null;this.drawOverlay();this.updateReadout(null);};

  refStack=(el)=>{this.cv.stackEl=el;if(el){if(!this._ro){this._ro=new ResizeObserver(()=>this.scheduleDraw());this._ro.observe(el);}[0,80,250,600].forEach(t=>setTimeout(()=>this.scheduleDraw(),t));}};
